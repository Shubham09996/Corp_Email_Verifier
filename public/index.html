API Integration Guide for Your 3-4 Projects

You can integrate this API directly into any of your Node.js, Python, PHP, or frontend microservices.

1. Verify Single Email (POST /api/verify)

Endpoint: POST http://localhost:3000/api/verify (Alias: POST /api/verify-corporate-email)

import axios from 'axios';

async function verifyCorporateEmail(email) {
  const response = await axios.post('http://localhost:3000/api/verify', {
    email: email,
    applicantId: 'USR_12345',
    options: {
      checkSmtp: true,
      checkDomainAge: true,
      allowFreeDomains: false // Reject Gmail/Yahoo/Hotmail
    }
  });

  const { status, isDeliverable, isCorporate, details } = response.data.data;
  console.log('Status:', status); // VALID | CATCH_ALL | PROTECTED | INVALID
  return response.data;
}
const response = await fetch('http://localhost:3000/api/verify', {

import requests

def verify_email(email):
    url = "http://localhost:3000/api/verify"
    payload = {
        "email": email,
        "options": {
            "allowFreeDomains": False,
            "checkSmtp": True
        }
    }
    response = requests.post(url, json=payload)
    return response.json()

result = verify_email("employee@stripe.com")
print(result)

  }'
